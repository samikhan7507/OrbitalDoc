cmake_minimum_required(VERSION 3.13)
project(OrbitalDock3D CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT EMSCRIPTEN)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(SRC src/main.cpp src/hud.cpp src/engine/renderer.cpp src/engine/gfx.cpp)
add_executable(orbitaldock ${SRC})
target_include_directories(orbitaldock PRIVATE src)
if(EMSCRIPTEN)
  set_target_properties(orbitaldock PROPERTIES OUTPUT_NAME index SUFFIX ".html" LINK_FLAGS
    "-sUSE_GLFW=3 -sMIN_WEBGL_VERSION=2 -sMAX_WEBGL_VERSION=2 -sALLOW_MEMORY_GROWTH=1 -sINITIAL_MEMORY=64MB -sENVIRONMENT=web -sEXPORTED_FUNCTIONS=_main,_set_touch_input,_ui_action,_orbit_drag,_orbit_zoom --shell-file ${CMAKE_SOURCE_DIR}/web/shell.html")
else()
  find_package(glfw3 QUIET)
  if(NOT glfw3_FOUND)           # Windows/others: download and build GLFW automatically
    include(FetchContent)
    set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git GIT_TAG 3.4)
    FetchContent_MakeAvailable(glfw)
  endif()
  target_link_libraries(orbitaldock PRIVATE glfw ${CMAKE_DL_LIBS})
endif()
